module mux_new(i,s,y);
	input [3:0]i;
	input [1:0]s;
	output reg y;
	
	always@(*)
	begin
		y = i[s];
	end
endmodule

module mux4_1_tb();

   //Step1 : Write down the variables required for testbench

   	reg [3:0]data_in;
	reg [1:0]sel_in;
	wire y_out;
	integer m,n;	
								
   //Step2 : Instantiate the Design 

   	mux4_1 DUT(data_in,sel_in,y_out);


   //Step3 : Declare a task to initialize inputs of DUT to 0 
   	task initialize;
		begin
			{data_in,sel_in} = 0;
		end
	endtask

   //Step4 : Declare  tasks with arguments for driving stimulus to DUT 
   	task sels(input [1:0]j);
		sel_in = j;
	endtask

	task inputs(input [3:0]k);
		data_in = k;
	endtask

	task delay();
		begin
			#10;
		end
	endtask


   //Step5 : Call the tasks from procedural process 
   

  	initial
	begin
		initialize;
		for(m = 0;m<4;m=m+1)
		begin
			sels(m);
			for(n=0;n<16;n=n+1)
			begin
				inputs(n);
				delay;
			end
		end


	$finish;
end


   //Step6 : Use $monitor task to display inputs and outputs

   initial
	   $monitor("data_in = %b,sel_in = %b,y_out = %b",data_in,sel_in,y_out);
  


   //Step7 : Use $finish task to terminate the simulation at 100ns
   
endmodule
