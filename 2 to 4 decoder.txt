2 to 4 decoder:

module decoder_2_4(en,d,y);
	input [1:0]d;
	input en;
	output [3:0]y;

	assign y[0] = ~d[1] & ~d[0] & en;
	assign y[1] = ~d[1] & d[0] & en;
	assign y[2] = d[1] & ~d[0] & en;
	assign y[3] = d[1] & d[0] & en;

endmodule
