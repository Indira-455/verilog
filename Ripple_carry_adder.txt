Full Adder:

module full_adder_bit(a,b,cin,sum,carry);
	input a,b,cin;
	output sum,carry;
	assign sum = a ^ b ^ cin;
	assign carry = a & b | b & cin | cin & a;

endmodule

Ripple Carry Adder:

module rca(a,b,cin,sum,carry);
	input [3:0]a,b;
	input cin;
	output [3:0]sum;
	output carry;
	wire w1,w2,w3;

		full_adder_bit FA1(a[0],b[0],cin,sum[0],w1);
		full_adder_bit FA2(a[1],b[1],w1,sum[1],w2);
		full_adder_bit FA3(a[2],b[2],w2,sum[2],w3);
		full_adder_bit FA4(a[3],b[3],w3,sum[3],carry);

endmodule
